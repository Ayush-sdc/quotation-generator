\newcommand{\addprojecttolist}[1]{%
    \xifinlist{#1}{\projectlist}{}{%
        \listgadd{\projectlist}{#1}%
    }%
}
% Define work item
\newcommand{\definework}[4]{%
    \addprojecttolist{#1}%
    \@ifundefined{proj:\detokenize{#1}}{%
        \expandafter\def\csname proj:\detokenize{#1}\endcsname{}%
        \expandafter\def\csname projtotal:\detokenize{#1}\endcsname{0}%
    }{}%
    \expandafter\g@addto@macro\csname proj:\detokenize{#1}\endcsname{%
        #2 & #3 & #4 & \fpeval{#3 * #4} \\
    }%
    \expandafter\xdef\csname projtotal:\detokenize{#1}\endcsname{\fpeval{\csname projtotal:\detokenize{#1}\endcsname + (#3 * #4)}}%
    \xdef\grandtotal{\fpeval{\grandtotal + (#3 * #4)}}%
}
% Define expense item
\newcommand{\defineexpense}[3]{%
    \addprojecttolist{#1}%
    \@ifundefined{proj:\detokenize{#1}}{%
        \expandafter\def\csname proj:\detokenize{#1}\endcsname{}%
        \expandafter\def\csname projtotal:\detokenize{#1}\endcsname{0}%
    }{}%
    \expandafter\g@addto@macro\csname proj:\detokenize{#1}\endcsname{%
        #2 & {---} & {---} & #3 \\
    }%
    \expandafter\xdef\csname projtotal:\detokenize{#1}\endcsname{\fpeval{\csname projtotal:\detokenize{#1}\endcsname + #3}}%
    \xdef\grandtotal{\fpeval{\grandtotal + #3}}%
}
